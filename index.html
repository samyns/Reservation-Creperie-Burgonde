// ============================================================ // Crêperie Burgonde — Google Apps Script v8 // - Compteur par service (midi/soir) et non par créneau // - Blocage manuel via onglet "Blocages" // - MAX 15 couverts par service // ============================================================ var EMAIL_GERANT = 'creperieburgonde@gmail.com'; var NOM_RESTAURANT = 'Crêperie Burgonde'; var TEL_RESTAURANT = ''; var MAX_COUVERTS = 15; // ------------------------------------------------------------ // Horaires fixes par jour // Retourne { midi: ["12h00","12h30",...], soir: ["19h00",...] } // Les clés absentes = service fermé ce jour // ------------------------------------------------------------ function getServicesJour(dateStr) { var date = new Date(dateStr + 'T12:00:00'); var jour = date.getDay(); // 0=dim,1=lun,2=mar,3=mer,4=jeu,5=ven,6=sam function slots(debut, fin) { var result = []; var parts = debut.split(':'); var h = parseInt(parts[0]), m = parseInt(parts[1]); var fparts = fin.split(':'); var hf = parseInt(fparts[0]), mf = parseInt(fparts[1]); while (h < hf || (h === hf && m <= mf)) { result.push(h + 'h' + (m === 0 ? '00' : String(m))); m += 30; if (m >= 60) { h++; m -= 60; } } return result; } var midi = slots('12:00', '13:30'); if (jour === 4) return {}; // jeudi fermé if (jour === 3) return { midi: midi }; // mercredi : midi seulement if (jour === 5) return { soir: slots('19:00', '22:00') }; // vendredi : soir seulement if (jour === 6) return { midi: midi, soir: slots('19:00', '22:00') }; // samedi // lundi, mardi, dimanche return { midi: midi, soir: slots('19:00', '21:00') }; } // Retourne le nom du service ("midi" ou "soir") pour un créneau donné function getServiceDuCreneau(heureStr) { var match = String(heureStr).match(/(\d+)h/); if (!match) return null; var h = parseInt(match[1]); return h < 15 ? 'midi' : 'soir'; } // Normalise une heure venant du Sheet (parfois convertie en Date par Sheets) function normaliserHeure(rawHeure) { if (rawHeure instanceof Date) { var h = rawHeure.getHours(); var m = rawHeure.getMinutes(); return h + 'h' + (m === 0 ? '00' : String(m)); } return String(rawHeure).trim(); } // ------------------------------------------------------------ // Lit les blocages manuels depuis l'onglet "Blocages" // Retourne un objet { "2026-10-18_soir": true, ... } // ------------------------------------------------------------ function getBlocages() { var blocages = {}; try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName('Blocages'); if (!sheet) return blocages; var lastRow = sheet.getLastRow(); if (lastRow <= 1) return blocages; var rows = sheet.getRange(2, 1, lastRow - 1, 2).getValues(); rows.forEach(function(row) { try { var rawDate = row[0]; var rawService = String(row[1]).trim().toLowerCase(); if (!rawDate || !rawService) return; var dateStr = rawDate instanceof Date ? Utilities.formatDate(rawDate, 'Europe/Paris', 'yyyy-MM-dd') : String(rawDate).trim(); if (rawService === 'midi' || rawService === 'soir') { blocages[dateStr + '_' + rawService] = true; } else if (rawService === 'journée entière') { blocages[dateStr + '_midi'] = true; blocages[dateStr + '_soir'] = true; } } catch(err) {} }); } catch(err) {} return blocages; } // ------------------------------------------------------------ // Compte les couverts par service pour une date // ------------------------------------------------------------ function getCouverstParService(dateStr) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var lastRow = sheet.getLastRow(); var compteur = { midi: 0, soir: 0 }; if (lastRow <= 1) return compteur; var data = sheet.getRange(2, 1, lastRow - 1, 10).getValues(); data.forEach(function(row) { try { var rowDate = row[5]; var rowDateStr = rowDate instanceof Date ? Utilities.formatDate(rowDate, 'Europe/Paris', 'yyyy-MM-dd') : String(rowDate).trim(); if (rowDateStr !== dateStr) return; var heure = normaliserHeure(row[6]); var service = getServiceDuCreneau(heure); if (!service) return; var nb = parseInt(String(row[7]).replace(/[^0-9]/g, '')) || 0; compteur[service] += nb; } catch(err) {} }); return compteur; } // ------------------------------------------------------------ // Point d'entrée unique — tout passe par doGet // ------------------------------------------------------------ function doGet(e) { var action = e.parameter.action; if (action === 'creneaux') return getDisponibilites(e.parameter.date); if (action === 'reserver') return traiterReservation(e.parameter.data); return ContentService .createTextOutput(JSON.stringify({ error: 'action inconnue' })) .setMimeType(ContentService.MimeType.JSON); } // ------------------------------------------------------------ // Créneaux disponibles pour une date // Retourne slots avec statut : dispo / complet / bloque // ------------------------------------------------------------ function getDisponibilites(dateStr) { if (!dateStr) { return ContentService .createTextOutput(JSON.stringify({ error: 'date manquante' })) .setMimeType(ContentService.MimeType.JSON); } var services = getServicesJour(dateStr); var blocages = getBlocages(); var couvertsPris = getCouverstParService(dateStr); // Construire la liste des slots avec leur disponibilité var slots = []; var pris = {}; // couverts pris par service (pour le front) ['midi', 'soir'].forEach(function(service) { if (!services[service]) return; var bloque = blocages[dateStr + '_' + service] || false; var couvertsPrisService = couvertsPris[service] || 0; pris[service] = couvertsPrisService; services[service].forEach(function(slot) { slots.push({ heure: slot, service: service, bloque: bloque, pris: couvertsPrisService }); }); }); return ContentService .createTextOutput(JSON.stringify({ slots: slots, max: MAX_COUVERTS })) .setMimeType(ContentService.MimeType.JSON); } // ------------------------------------------------------------ // Traitement d'une réservation // ------------------------------------------------------------ function traiterReservation(rawData) { try { var data = JSON.parse(rawData); var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow(['Date soumission', 'Prénom', 'Nom', 'Email', 'Téléphone', 'Date réservation', 'Heure', 'Couverts', 'Allergies', 'Message']); sheet.getRange(1, 1, 1, 10).setFontWeight('bold'); sheet.setFrozenRows(1); } // Vérif blocage var blocages = getBlocages(); var service = getServiceDuCreneau(data.heure); if (blocages[data.date + '_' + service]) { return ContentService .createTextOutput(JSON.stringify({ error: 'bloque' })) .setMimeType(ContentService.MimeType.JSON); } // Vérif capacité par service var couvertsPris = getCouverstParService(data.date); var dejaPris = couvertsPris[service] || 0; var nbDemandes = parseInt(String(data.couverts).replace(/[^0-9]/g, '')) || 1; if (dejaPris + nbDemandes > MAX_COUVERTS) { return ContentService .createTextOutput(JSON.stringify({ error: 'complet' })) .setMimeType(ContentService.MimeType.JSON); } try { insererChronologique(sheet, data); } catch(insertErr) { Logger.log('insererChronologique planté : ' + insertErr.message); sheet.appendRow([ data.soumis_le, data.prenom, data.nom, data.email, data.tel, data.date, data.heure, data.couverts, data.allergie || '', data.message || '' ]); } var dateObj = new Date(data.date + 'T12:00:00'); var dateFormatee = dateObj.toLocaleDateString('fr-FR', { weekday: 'long', day: 'numeric', month: 'long', year: 'numeric' }); envoyerConfirmationClient(data, dateFormatee); envoyerNotificationGerant(data, dateFormatee); return ContentService .createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } catch(err) { Logger.log('traiterReservation erreur : ' + err.message); return ContentService .createTextOutput(JSON.stringify({ error: err.message })) .setMimeType(ContentService.MimeType.JSON); } } // ------------------------------------------------------------ // Insertion chronologique // ------------------------------------------------------------ function heureEnMinutes(heureStr) { var match = String(heureStr).match(/(\d+)h(\d*)/); if (!match) return 0; return parseInt(match[1]) * 60 + (parseInt(match[2]) || 0); } function insererChronologique(sheet, data) { var lastRow = sheet.getLastRow(); var newRow = [ data.soumis_le, data.prenom, data.nom, data.email, data.tel, data.date, data.heure, data.couverts, data.allergie || '', data.message || '' ]; if (lastRow <= 1) { sheet.appendRow(newRow); return; } var rows = sheet.getRange(2, 1, lastRow - 1, 10).getValues(); var insertAt = lastRow + 1; for (var i = 0; i < rows.length; i++) { try { var rowDate = rows[i][5]; var rowDateStr = rowDate instanceof Date ? Utilities.formatDate(rowDate, 'Europe/Paris', 'yyyy-MM-dd') : String(rowDate).trim(); var rowHeure = heureEnMinutes(normaliserHeure(rows[i][6])); if (rowDateStr > data.date || (rowDateStr === data.date && rowHeure > heureEnMinutes(data.heure))) { insertAt = i + 2; break; } } catch(err) { continue; } } if (insertAt > lastRow) { sheet.appendRow(newRow); } else { sheet.insertRowBefore(insertAt); sheet.getRange(insertAt, 1, 1, 10).setValues([newRow]); } } // ------------------------------------------------------------ // Suppression horaire des créneaux passés // ------------------------------------------------------------ function supprimerReservationsPassees() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var lastRow = sheet.getLastRow(); if (lastRow <= 1) return; var maintenant = new Date(); var rows = sheet.getRange(2, 1, lastRow - 1, 10).getValues(); for (var i = rows.length - 1; i >= 0; i--) { try { var rowDate = rows[i][5]; var dateStr = rowDate instanceof Date ? Utilities.formatDate(rowDate, 'Europe/Paris', 'yyyy-MM-dd') : String(rowDate).trim(); if (!dateStr) continue; var heureNorm = normaliserHeure(rows[i][6]); var match = heureNorm.match(/(\d+)h(\d*)/); if (!match) continue; var parts = dateStr.split('-').map(Number); var creneauDate = new Date(parts[0], parts[1]-1, parts[2], parseInt(match[1]), parseInt(match[2])||0, 0); if (creneauDate < maintenant) sheet.deleteRow(i + 2); } catch(err) { continue; } } } // ------------------------------------------------------------ // Créer l'onglet Blocages avec liste déroulante (lancer une fois) // ------------------------------------------------------------ function creerOngletBlocages() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var existing = ss.getSheetByName('Blocages'); if (existing) { Logger.log('Onglet Blocages existe déjà'); return; } var sheet = ss.insertSheet('Blocages'); // En-têtes sheet.getRange('A1').setValue('Date'); sheet.getRange('B1').setValue('Service'); sheet.getRange(1, 1, 1, 2).setFontWeight('bold'); sheet.setFrozenRows(1); // Liste déroulante sur la colonne B var rule = SpreadsheetApp.newDataValidation() .requireValueInList(['Midi', 'Soir', 'Journée entière'], true) .setAllowInvalid(false) .build(); sheet.getRange('B2:B100').setDataValidation(rule); // Format date sur la colonne A sheet.getRange('A2:A100').setNumberFormat('yyyy-MM-dd'); // Largeur des colonnes sheet.setColumnWidth(1, 130); sheet.setColumnWidth(2, 160); Logger.log('Onglet Blocages créé avec succès'); } // ------------------------------------------------------------ // Installer le trigger horaire — lancer UNE SEULE FOIS // ------------------------------------------------------------ function installerTrigger() { ScriptApp.getProjectTriggers().forEach(function(t) { if (t.getHandlerFunction() === 'supprimerReservationsPassees') ScriptApp.deleteTrigger(t); }); ScriptApp.newTrigger('supprimerReservationsPassees').timeBased().everyHours(1).create(); } // ------------------------------------------------------------ // Email confirmation client // ------------------------------------------------------------ function envoyerConfirmationClient(data, dateFormatee) { var sujet = 'Votre réservation — ' + NOM_RESTAURANT; var telLigne = TEL_RESTAURANT ? '
